31[[nodiscard]] AABB clampedBounds(const Vector2& center, const Vector2& halfExtents) {
32 Vector2 clampedCenter = center;
33 clampedCenter.x = std::clamp(clampedCenter.x, kWorldMin + halfExtents.x, kWorldMax - halfExtents.x);

131void World::worldToGrid(const Vector2& position, int& gridX, int& gridY) const {
132 if (m_cellWidth <= 0.f || m_cellHeight <= 0.f) {
133 gridX = 0;
134 gridY = 0;

137
138 gridX = static_cast<int>(std::floor((position.x - kWorldMin) / m_cellWidth));
139 gridY = static_cast<int>(std::floor((position.y - kWorldMin) / m_cellHeight));
140 gridX = std::clamp(gridX, 0, std::max(0, m_gridWidth - 1));
141 gridY = std::clamp(gridY, 0, std::max(0, m_gridHeight - 1));

183void World::generateArena(int gridWidth, int gridHeight, float emptyCellChance) {
184 clear();
185
186 m_gridWidth = gridWidth;
187 m_gridHeight = gridHeight;
188 m_cellWidth = 2.f / static_cast<float>(gridWidth);
189 m_cellHeight = 2.f / static_cast<float>(gridHeight);
190
191 const Vector2 halfExtents = getCellHalfExtents();
192 auto& random = Random::getInstance();
193
194 for (int gridY = 0; gridY < gridHeight; ++gridY) {
195 for (int gridX = 0; gridX < gridWidth; ++gridX) {
196 const Vector2 center = gridToWorld(gridX, gridY);
197 const bool isBorder = gridX == 0 || gridY == 0 || gridX == gridWidth - 1 ||
198 gridY == gridHeight - 1;
199 const bool isPillar = gridX % 2 == 0 && gridY % 2 == 0;
200
201 if (isBorder || isPillar) {
202 addEntity(createWall(center, halfExtents, true));
203 continue;
204 }
205
206 if (isSpawnCell(gridX, gridY, gridWidth, gridHeight)) {
207 continue;
208 }
209
210 if (random.range(0.f, 1.f) < emptyCellChance) {
211 continue;
212 }
213
214 addEntity(createWall(center, halfExtents, false));
215 }
216 }

336bool World::isPositionBlocked(const AABB& bounds, unsigned int moverId) const {
337 for (const auto& entity : m_entities) {
338 if (!entity || !entity->isActive() || entity->getId() == moverId) {
339 continue;
340 }
341
342 if (!entity->getBounds().intersects(bounds)) {
343 continue;
344 }
345
346 switch (entity->getType()) {
347 case EntityType::Wall:
348 return true;
349 case EntityType::Bomb: {
350 const auto bomb = std::static_pointer_cast<Bomb>(entity);
351 if (bomb->isExploded()) {
352 break;
353 }
354 if (!canPassBomb(moverId, bomb->getId())) {
355 return true;
356 }
357 break;
358 }
359 case EntityType::Character: {
360 const auto character = std::static_pointer_cast<Character>(entity);
361 if (character->isAlive()) {
362 return true;
363 }
364 break;
365 }
366 case EntityType::PowerUp:
367 break;
368 }
369 }
370
371 return false;

402bool World::tryMoveCharacter(unsigned int characterId, const Vector2& direction,
403 float deltaTime) {
404 if (!isPlaying()) {
405 return false;
406 }
407
408 const std::shared_ptr<Character> character = findCharacter(characterId);
409 if (!character || !character->isAlive()) {
410 return false;
411 }
412
413 const Vector2 normalizedDirection = direction.normalized();
414 if (normalizedDirection.lengthSquared() <= 0.f) {
415 return false;
416 }
417
418 const Vector2 delta = normalizedDirection * (character->getMoveSpeed() * deltaTime);
419 const Vector2 targetPosition = character->getPosition() + delta;
420 const AABB targetBounds =
421 clampedBounds(targetPosition, character->getHalfExtents());
422
423 if (isPositionBlocked(targetBounds, characterId)) {
424 return false;
425 }
426
427 character->setPosition(targetBounds.getCenter());
428 tryPickupPowerUp(character);
429 return true;
430}

543void World::castExplosionRay(const Vector2& origin, const Vector2& direction, int maxRadius,
544 unsigned int ownerId) {
545 if (maxRadius <= 0) {
546 return;
547 }
548
549 int originX = 0;
550 int originY = 0;
551 worldToGrid(origin, originX, originY);
552
553 const int stepX = static_cast<int>(direction.x);
554 const int stepY = static_cast<int>(direction.y);
555
556 for (int radius = 1; radius <= maxRadius; ++radius) {
557 const int gridX = originX + stepX * radius;
558 const int gridY = originY + stepY * radius;
559
560 if (gridX < 0 || gridY < 0 || gridX >= m_gridWidth || gridY >= m_gridHeight) {
561 return;
562 }
563
564 const Vector2 cellCenter = gridToWorld(gridX, gridY);
565 bool blocked = false;
566
567 for (const auto& entity : m_entities) {
568 if (!entity || !entity->isActive()) {
569 continue;
570 }
571
572 int entityX = 0;
573 int entityY = 0;
574 worldToGrid(entity->getPosition(), entityX, entityY);
575 if (entityX != gridX || entityY != gridY) {
576 continue;
577 }
578
579 switch (entity->getType()) {
580 case EntityType::Wall: {
581 const auto wall = std::static_pointer_cast<Wall>(entity);
582 if (wall->isIndestructible()) {
583 return;
584 }
585
586 const unsigned int wallId = entity->getId();
587 entity->setActive(false);
588 removeEntity(wallId);
589 dispatchWorldEvent(GameEventType::BlockDestroyed, wallId, ownerId);
590
591 if (Random::getInstance().range(0.f, 1.f) < kPowerUpDropChance) {
592 const int roll = Random::getInstance().range(0, 2);
593 const PowerUpType type = static_cast<PowerUpType>(roll);
594 queueSpawn(createPowerUp(cellCenter, getPowerUpHalfExtents(), type));
595 }
596
597 blocked = true;
598 break;
599 }
600 case EntityType::Bomb: {
601 const auto bomb = std::static_pointer_cast<Bomb>(entity);
602 if (!bomb->isExploded()) {
603 m_detonationQueue.push_back(bomb->getId());
604 }
605 blocked = true;
606 break;
607 }
608 case EntityType::Character:
609 break;
610 case EntityType::PowerUp:
611 entity->setActive(false);
612 removeEntity(entity->getId());
613 break;
614 }
615 }
616
617 killCharactersInBlastCell(cellCenter, ownerId);
618
619 if (blocked) {
620 return;
621 }
622 }

625void World::detonateBomb(const std::shared_ptr<Bomb>& bomb) {
626 if (!bomb || bomb->isExploded()) {
627 return;
628 }
629
630 bomb->setExploded(true);
631 bomb->setActive(false);
632 removeEntity(bomb->getId());
633
634 const Vector2 origin = snapToCellCenter(bomb->getPosition());
635 const unsigned int ownerId = bomb->getOwnerId();
636
637 int blastRadius = std::max(1, bomb->getBlastRadius());
638 if (const std::shared_ptr<Character> owner = findCharacter(ownerId)) {
639 blastRadius = std::max(1, owner->getBlastRadius());
640 }
641
642 killCharactersInBlastCell(origin, ownerId);
643
644 castExplosionRay(origin, kDirectionRight, blastRadius, ownerId);
645 castExplosionRay(origin, kDirectionLeft, blastRadius, ownerId);
646 castExplosionRay(origin, kDirectionUp, blastRadius, ownerId);
647 castExplosionRay(origin, kDirectionDown, blastRadius, ownerId);

698void World::update(float deltaTime) {
699 if (!isPlaying()) {
700 return;
701 }
702
703 updateBombExitState();
704 updateBombs(deltaTime);
705
706 m_detonationScratch->assign(m_detonationQueue.begin(), m_detonationQueue.end());
707 m_detonationQueue.clear();
708 std::sort(m_detonationScratch->begin(), m_detonationScratch->end());
709 m_detonationScratch->erase(
710 std::unique(m_detonationScratch->begin(), m_detonationScratch->end()),
711 m_detonationScratch->end());
712
713 for (const unsigned int bombId : *m_detonationScratch) {
714 const std::shared_ptr<Entity> entity = findEntity(bombId);
715 if (!entity || entity->getType() != EntityType::Bomb) {
716 continue;
717 }
718 detonateBomb(std::static_pointer_cast<Bomb>(entity));
719 }
720
721 flushPendingChanges();
722 evaluateOutcome();
723 notifyEntityTicks();
724 dispatchWorldEvent(GameEventType::WorldTick);
725}
