195CharacterView::FacingDirection CharacterView::directionFromDelta(const sf::Vector2f& delta) const {
196 const float absX = std::abs(delta.x);
197 const float absY = std::abs(delta.y);
198
199 if (absX >= absY) {
200 return delta.x >= 0.f ? FacingDirection::Right : FacingDirection::Left;
201 }
202
203 return delta.y >= 0.f ? FacingDirection::Down : FacingDirection::Up;
204}

206void CharacterView::configurePlayerSprite(const sf::Vector2f& size, const sf::Vector2f& center,
207 FacingDirection facing, std::size_t frameIndex) {
208 const sf::IntRect rect = [&]() {
209 switch (facing) {
210 case FacingDirection::Down:
211 return kPlayerDown[frameIndex % kPlayerDown.size()];
212 case FacingDirection::Up:
213 return kPlayerUp[frameIndex % kPlayerUp.size()];
214 case FacingDirection::Right:
215 case FacingDirection::Left:
216 return kPlayerRight[frameIndex % kPlayerRight.size()];
217 }
218
219 return kPlayerDown[0];
220 }();
221
222 const bool flipHorizontally = facing == FacingDirection::Left;
223
224 m_sprite.setTexture(spriteAtlas(), true);
225 m_sprite.setTextureRect(rect);
226 m_sprite.setOrigin(static_cast<float>(rect.width) * 0.5f, static_cast<float>(rect.height) * 0.5f);
227 m_sprite.setScale((flipHorizontally ? -1.f : 1.f) * size.x / static_cast<float>(rect.width),
228 size.y / static_cast<float>(rect.height));
229 m_sprite.setPosition(center);
230 m_sprite.setColor(sf::Color(255, 255, 255));
231
232 m_facing = facing;
233 m_frameIndex = frameIndex;
234}

236void CharacterView::syncFromModel() {
237 if (!hasSpriteAtlas()) {
238 m_visible = false;
239 return;
240 }
241
242 const auto model = lockModel();
243 if (!model || !model->isActive()) {
244 m_visible = false;
245 return;
246 }
247
248 const auto character = std::static_pointer_cast<Logic::Character>(model);
249 if (!character->isAlive()) {
250 m_visible = false;
251 return;
252 }
253
254 const sf::Vector2f size = toPixelSize(model->getHalfExtents() * 2.f);
255 const sf::Vector2f center = toPixels(model->getPosition());
256
257 if (!character->isPlayer()) {
258 configureSprite(m_sprite, kCharacterAltRect, size * kCharacterSpriteScale, center,
259 sf::Color(255, 210, 210));
260 m_visible = true;
261 return;
262 }
263
264 if (!m_hasLastPosition) {
265 m_lastPosition = center;
266 m_hasLastPosition = true;
267 configurePlayerSprite(size * kCharacterSpriteScale, center, FacingDirection::Down, 0);
268 m_visible = true;
269 return;
270 }
271
272 const sf::Vector2f delta = center - m_lastPosition;
273 const float movedDistanceSquared = delta.x * delta.x + delta.y * delta.y;
274 if (movedDistanceSquared > 0.0001f) {
275 const FacingDirection nextFacing = directionFromDelta(delta);
276 const std::size_t nextFrameIndex = nextFacing == m_facing
277 ? (m_frameIndex + 1)
278 : 0;
279
280 configurePlayerSprite(size * kCharacterSpriteScale, center, nextFacing, nextFrameIndex);
281 m_lastPosition = center;
282 m_visible = true;
283 return;
284 }
285
286 m_sprite.setPosition(center);
287 m_visible = true;
288}
