23bool isPositionThreatened(const Logic::World& world, const Logic::Vector2& position) {
24 for (const auto& entity : world.getEntities()) {
25 if (!entity || !entity->isActive() || entity->getType() != Logic::EntityType::Bomb) {
26 continue;
27 }
28
29 const auto bomb = std::static_pointer_cast<Logic::Bomb>(entity);
30 if (bomb->isExploded()) {
31 continue;
32 }
33
34 const Logic::Vector2 delta = position - bomb->getPosition();
35 const float blastReach = world.getCellWidth() * static_cast<float>(bomb->getBlastRadius() + 1);
36
37 if (delta.lengthSquared() < 0.22f) {
38 return true;
39 }
40
41 const bool sameRow = std::abs(delta.y) < 0.10f && std::abs(delta.x) <= blastReach;
42 const bool sameColumn = std::abs(delta.x) < 0.10f && std::abs(delta.y) <= blastReach;
43 if (sameRow || sameColumn) {
44 return true;
45 }
46 }
47
48 return false;
49}

146void PlayState::updateBots(float deltaTime) {
147 static std::unordered_map<unsigned int, float> botDecisionTimers;
148 static std::unordered_map<unsigned int, Logic::Vector2> botDirections;
149 static std::unordered_map<unsigned int, float> botBombCooldowns;
150
151 auto& random = Logic::Random::getInstance();
152 const bool botsMayBomb = m_matchTime >= 5.f;
153 const float moveStride = 6.0f;
154
155 for (const auto& entity : m_world.getEntities()) {
156 if (!entity || !entity->isActive() || entity->getType() != Logic::EntityType::Character) {
157 continue;
158 }
159
160 const auto character = std::static_pointer_cast<Logic::Character>(entity);
161 if (!character->isAlive() || character->isPlayer()) {
162 continue;
163 }
164
165 float& decisionTimer = botDecisionTimers[character->getId()];
166 decisionTimer -= deltaTime;
167 if (decisionTimer > 0.f) {
168 continue;
169 }
170
171 const auto player = m_world.findCharacter(m_playerId);
172 const bool currentlySafe = !isPositionThreatened(m_world, character->getPosition());
173
174 std::array<Logic::Vector2, 4> directions = {
175 Logic::Vector2{1.f, 0.f},
176 Logic::Vector2{-1.f, 0.f},
177 Logic::Vector2{0.f, 1.f},
178 Logic::Vector2{0.f, -1.f},
179 };
180
181 const int bias = static_cast<int>(character->getId()) % 4;
182 std::rotate(directions.begin(), directions.begin() + bias, directions.end());
183
184 if (player) {
185 const Logic::Vector2 toPlayer = player->getPosition() - character->getPosition();
186 if (std::abs(toPlayer.x) > std::abs(toPlayer.y)) {
187 directions[0] = {toPlayer.x >= 0.f ? 1.f : -1.f, 0.f};
188 directions[1] = {0.f, toPlayer.y >= 0.f ? 1.f : -1.f};
189 } else if (toPlayer.lengthSquared() > 0.f) {
190 directions[0] = {0.f, toPlayer.y >= 0.f ? 1.f : -1.f};
191 directions[1] = {toPlayer.x >= 0.f ? 1.f : -1.f, 0.f};
192 }
193 }
194
195 Logic::Vector2 bestDirection{};
196 float bestScore = std::numeric_limits<float>::infinity();
197
198 for (const auto& direction : directions) {
199 if (direction.lengthSquared() == 0.f) {
200 continue;
201 }
202
203 const Logic::Vector2 target = character->getPosition() + direction * 0.9f;
204 const bool safeTarget = !isPositionThreatened(m_world, target);
205 if (!safeTarget) {
206 continue;
207 }
208
209 float score = 0.f;
210 if (player) {
211 score += (player->getPosition() - target).lengthSquared();
212 }
213
214 const bool movingAwayFromBomb = !isPositionThreatened(m_world, target + direction * 0.25f);
215 if (!movingAwayFromBomb) {
216 score += 1000.f;
217 }
218
219 if (score < bestScore) {
220 bestScore = score;
221 bestDirection = direction;
222 }
223 }
224
225 bool moved = false;
226 if (bestDirection.lengthSquared() > 0.f) {
227 moved = m_world.tryMoveCharacter(character->getId(), bestDirection, deltaTime * moveStride);
228 }
229
230 if (moved) {
231 botDirections[character->getId()] = bestDirection;
232 } else {
233 botDirections.erase(character->getId());
234 decisionTimer = 0.12f + random.range(0.f, 0.17f);
235 continue;
236 }
237
238 float& bombCooldown = botBombCooldowns[character->getId()];
239 bombCooldown -= deltaTime;
240
241 if (botsMayBomb && bombCooldown <= 0.f && currentlySafe && player) {
242 const Logic::Vector2 toPlayer = player->getPosition() - character->getPosition();
243 const bool aligned = std::abs(toPlayer.x) < 0.25f || std::abs(toPlayer.y) < 0.25f;
244 const bool closeEnough = toPlayer.lengthSquared() < 1.8f;
245 const bool safeToBomb = !isPositionThreatened(m_world, character->getPosition());
246 const bool canHitPlayer = bombCanHitPlayer(m_world, *character, *player);
247
248 if (aligned && closeEnough && safeToBomb && canHitPlayer &&
249 m_world.tryPlaceBomb(character->getId())) {
250 bombCooldown = 1.8f + random.range(0.f, 1.4f) +
251 static_cast<float>((character->getId() % 3)) * 0.2f;
252 botDirections.erase(character->getId());
253 }
254 }
255
256 decisionTimer = 0.12f + random.range(0.f, 0.25f) +
257 static_cast<float>((character->getId() % 5)) * 0.05f;
258 }
