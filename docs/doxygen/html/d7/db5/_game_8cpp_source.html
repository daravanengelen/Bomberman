53void Game::run() {

55 stopwatch.reset();
56 changeState(std::make_unique<MenuState>());
57
58 while (m_window.isOpen()) {
59 stopwatch.tick();
60 const float deltaTime =

62
63 sf::Event event{};

65 if (m_currentState) {
66 m_currentState->handleEvent(*this, event);

68 }
69
70 if (m_currentState) {
